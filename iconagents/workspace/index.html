Quick Start

import { IconAgent } from './iconagents-sdk.mjs';

const agent = new IconAgent({
  apiKey: 'grk_your_key_here',
  rail: 'themusicindustry.ai'
});

// Start a rollout chat
const session = agent.startRollout({ artist: 'Your Artist' });
await session.send('Week 4 of my rollout');

// Get daily intel
const intel = await agent.getIntel({ date: '2026-07-24' });

// Check token balance
const balance = await agent.getBalance();
